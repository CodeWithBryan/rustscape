// Records the factory method and arguments of every projectile that is sent. A
// nine-argument create is a template: its own arguments (with its source and
// victim) are shown again when a copy or a transform of it is sent.
public class Projectile{
 final int id;final List<Object> made;final List<Object> template;
 public Projectile(int id){this.id=id;this.made=null;this.template=null;}
 Projectile(int id,List<Object> made,List<Object> template){this.id=id;this.made=made;this.template=template;}
 public static Projectile create(Entity s,Entity v,int id,int startHeight,int endHeight,int startDelay,int speed,int angle,int distance){return new Projectile(id,null,Arrays.asList(Support.tag(s),Support.tag(v),id,startHeight,endHeight,startDelay,speed,angle,distance));}
 public static Projectile create(Entity s,Entity v,int id,int startHeight,int endHeight,int startDelay,int speed,int angle){return new Projectile(id,Arrays.asList("create",Support.tag(s),Support.tag(v),id,startHeight,endHeight,startDelay,speed,angle),null);}

  if(template==null)return new Projectile(id,Arrays.asList("copy",id,Support.tag(s),Support.tag(v),Support.hex(speed)),null);
  List<Object> row=new ArrayList<>();row.add("copyTemplate");row.addAll(template);row.add(Support.tag(s));row.add(Support.tag(v));row.add(Support.hex(speed));return new Projectile(id,row,null);}

  if(template==null)throw Support.unsupported("Projectile.transform of a made projectile");
  List<Object> row=new ArrayList<>();row.add("transform");row.addAll(template);row.add(Support.tag(s));row.add(Support.tag(v));row.add(npc);row.add(baseSpeed);row.add(modifiedSpeed);return new Projectile(id,row,null);}
 public void send(){if(made==null)throw Support.unsupported("Projectile.send on a template");Support.emit("projectile",made.toArray());}

// The AbstractNPC constructors the dragon classes reach: the object is the case

 public AbstractNPC(int id,Location location){this(id,location,true);}
 public AbstractNPC(int id,Location location,boolean autowalk){super(Support.pendingTag,Support.pendingSpec);}
 public abstract AbstractNPC construct(int id,Location location,Object... objects);
 public abstract int[] getIds();
}

package core.api

fun getItemFromEquipment(player: Player, slot: EquipmentSlot): Item? = player.equipment.get(slot.ordinal)
fun getSlayerTask(player: Player): Tasks? { Support.emit("slayerTask", player.tag); return Tasks.of(player.spec["slayerTask"] as Map<*, *>?) }
fun inEquipment(player: Player, id: Int, amount: Int = 1): Boolean { Support.emit("inEquipment", player.tag, id, amount); return Support.has(player.spec, "inEquipment", id.toString()) }
fun applyPoison(entity: Entity, source: Entity, severity: Int) { Support.emit("poison", entity.tag, source.tag, severity) }
@JvmOverloads fun playGlobalAudio(location: Location, id: Int, delay: Int = 0, loops: Int = 1, radius: Int = 8) { Support.emit("audio", Support.coord(location), id, delay, loops, radius) }
fun playHurtAudio(player: Player, delay: Int = 0) { Support.emit("hurtAudio", player.tag, delay) }
fun log(origin: Class<*>, type: core.tools.Log, message: String) { Support.emit("log", origin.simpleName, type.name, message) }
// Audio.defaultAudioRadius is 8.
@JvmOverloads fun playAudio(player: Player, id: Int, delay: Int = 0, loops: Int = 1, location: Location? = null, radius: Int = 8) { Support.emit("playAudio", player.tag, id, delay, loops, Support.coord(location), radius) }
// ContentAPI sendMessage (it splits the text into lines for Player.sendMessages): recorded as one call.
fun sendMessage(player: Player, message: String) { Support.emit("apiMessage", player.tag, message) }
// setVarp writes the varp map and sends the packet: recorded as one call.
fun setVarp(player: Player, varpIndex: Int, value: Int, save: Boolean = false) { Support.emit("setVarp", player.tag, varpIndex, value, save) }
// delayEntity sets the script delay from GameWorld.ticks and locks the entity: recorded as one call.
fun delayEntity(entity: Entity, ticks: Int) { Support.emit("delayEntity", entity.tag, ticks) }
// replaceSlot: recorded; the model then holds the new item in that slot.
fun replaceSlot(player: Player, slot: Int, item: Item, currentItem: Item? = null, container: Container = Container.INVENTORY): Item? { Support.emit("replaceSlot", player.tag, slot, listOf(item.id, item.amount, item.charge), if (currentItem == null) null else listOf(currentItem.id, currentItem.amount, currentItem.charge), container.name); Support.replaceSlot(player, slot, item, container.name); return null }
