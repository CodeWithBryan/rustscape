 /*SLOT_CONSTANTS*/
 final Player p;final Map<Integer,Item> items=new HashMap<>();public EquipmentContainer(Player p){this.p=p;}

 Item at(int slot){Map m=(Map)p.spec.get("equipment");List v=m==null?null:(List)m.get(String.valueOf(slot));if(v==null)return null;return items.computeIfAbsent(slot,k->new Item(p.tag+":"+slot,Support.n(v.get(0)),Support.n(v.get(1)),Support.n(v.get(2)),(String)v.get(3),v.size()>4?v.get(4):null));}
 public Item get(int slot){Support.emit("equip",p.tag,slot);if(slot<0||slot>=14)return null;return at(slot);}
 public Item getNew(int slot){Support.emit("equipNew",p.tag,slot);if(slot<0||slot>=14)throw new ArrayIndexOutOfBoundsException(slot);Item i=at(slot);return i!=null?i:new Item(0);}
 public boolean containsAtLeastOneItem(int id){Support.emit("containsOne",p.tag,id);return Support.has(p.spec,"contains",String.valueOf(id));}
 public boolean contains(int id,int amount){throw Support.unsupported("Container.contains");}
